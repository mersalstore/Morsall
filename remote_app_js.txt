const { createServer } = require('http');
const { parse } = require('url');
const path = require('path');
const fs = require('fs');

// Log to file in public_html for now
const logFile = fs.createWriteStream(path.join(__dirname, 'server.log'), { flags: 'a' });
process.stdout.write = process.stderr.write = logFile.write.bind(logFile);

console.log('--- RADICAL RESTART STARTING ---');
console.log('Time:', new Date().toISOString());

// Point to nodejs folder for node_modules and binary
const nodejsDir = '/home/u754458241/domains/morsall.com/nodejs';
const nodeModulesPath = path.join(nodejsDir, 'node_modules');
const prismaBinary = path.join(nodeModulesPath, '.prisma/client/query-engine-rhel-openssl-3.0.x');

process.env.PRISMA_CLIENT_ENGINE_TYPE = 'binary';
process.env.PRISMA_QUERY_ENGINE_BINARY = prismaBinary;
process.env.NODE_ENV = 'production';

// Ensure binary exists and is executable
if (fs.existsSync(prismaBinary)) {
    try { fs.chmodSync(prismaBinary, '755'); } catch(e) {}
} else {
    console.error('MISSING BINARY AT: ' + prismaBinary);
}

// Setup paths
process.env.NODE_PATH = [
  nodeModulesPath,
  path.join(nodeModulesPath, 'next/node_modules'),
  path.join(nodeModulesPath, '@next/env/node_modules')
].join(path.delimiter);
require('module').Module._initPaths();

const next = require('next');
const app = next({ dev: false, dir: nodejsDir });
const handle = app.getRequestHandler();

app.prepare().then(() => {
  createServer(async (req, res) => {
    try {
      const parsedUrl = parse(req.url, true);
      const { pathname } = parsedUrl;
      
      if (pathname === '/diag') {
        res.setHeader('Content-Type', 'application/json');
        res.end(JSON.stringify({
          status: 'RADICAL_RESTART_ACTIVE',
          binary: fs.existsSync(prismaBinary),
          env: process.env.PRISMA_CLIENT_ENGINE_TYPE
        }));
        return;
      }

      await handle(req, res, parsedUrl);
    } catch (err) {
      console.error('SERVER_ERROR:', err);
      res.statusCode = 500;
      res.end('Internal Server Error');
    }
  }).listen(process.env.PORT || 3000, () => {
      console.log('Radical Server listening on:', process.env.PORT || 3000);
  });
}).catch(err => {
  console.error('PREPARE_ERROR:', err);
  process.exit(1);
});
